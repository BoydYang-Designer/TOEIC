/* ===== 核心多益單字的缺口（還沒入庫的）：none＝完全沒有詳解、gap＝有詳解但缺內容。已入庫的字交給下面一般清單處理，不重複列 ===== */
function dnCoreGaps(kind) {
  const seen = new Set(), out = [];
  dnArts().forEach(x => {
    const id = idOf(x), list = [];
    (x.vocab || []).forEach((v, i) => {
      const t = String((v && v.word) || '').trim(), k = dnKey(t); if (!t || seen.has(k)) return; seen.add(k);
      if (dnMarks().some(m => dnKey(m.text) === k)) return;
      const has = !!dnExtraSrc(x, t); if (kind === 'none' ? has : !has) return;
      const gaps = has ? dnGapsOf({ id: '~', qid: id, text: t, type: /\s/.test(t) ? 'phrase' : 'word' }) : [];
      if (kind === 'gap' && !gaps.length) return;
      list.push({ v, t, i, gaps });
    });
    if (list.length) out.push({ x, id, list });
  });
  return out;
}

function dnImportMarks(input) {
  const f = input.files[0]; if (!f) return; const fr = new FileReader();
  fr.onload = () => {
    try {
      const j = JSON.parse(fr.result), arr = Array.isArray(j) ? j : j && j.marks;
      if (!Array.isArray(arr)) throw new Error('找不到 marks 陣列');
      let add = 0, upd = 0;
      arr.forEach(r => {
        if (!r || typeof r.id !== 'string' || typeof r.qid !== 'string' || !r.text) return;
        const n = { id: r.id, qid: r.qid, text: String(r.text), type: DNT[r.type] ? r.type : 'word', ctx: String(r.ctx || r.text), from: +r.from || 0, to: +r.to || 0, note: String(r.note || ''), ...(r.nw ? { nw: String(r.nw) } : {}), created: +r.created || Date.now(), updated: +r.updated || +r.created || Date.now() };
        const o = dnMarks().find(q => q.id === n.id) || dnMarks().find(q => q.qid === n.qid && dnKey(q.text) === dnKey(n.text));
        if (!o) { dnMarks().push(n); add++; } else if (n.updated > (o.updated || 0)) { n.id = o.id; Object.assign(o, n); upd++; }
      });
      save(); dnToast(`匯入完成：新增 ${add}、更新 ${upd}`); render();
    } catch (e) { alert('匯入失敗：' + e.message); }
  };
  fr.readAsText(f, 'utf-8'); input.value = '';
}

# ------------------------------- 共用小工具 -------------------------------
def ok_file(p: Path) -> bool:
    return p.exists() and p.stat().st_size > 1000


# ------------------------------- 版本紀錄（voices-log.json） -------------------------------
# STATE["redo"]：0＝只補缺的檔；1＝再加上「紀錄的聲音／語速／引擎和目前設定不同」的檔；2＝再加上「沒有紀錄的舊檔」
STATE = dict(root=None, log={}, redo=0, dirty=0)


def log_path(root) -> Path:
    return Path(root) / LOG_NAME


def begin(root):
    """讀取版本紀錄（每個 plan_* 開頭呼叫）。"""
    root = Path(root)
    STATE["root"] = root
    try:
        j = json.loads(log_path(root).read_text(encoding="utf-8-sig"))
        STATE["log"] = j.get("files") or {} if isinstance(j, dict) else {}
    except (OSError, ValueError):
        STATE["log"] = {}
    STATE["dirty"] = 0


def rel_key(path):
    try:
        return Path(path).relative_to(STATE["root"]).as_posix()
    except (ValueError, TypeError):
        return Path(path).as_posix()


def need(path, sig, rate):
    """回傳 None＝不用做；"new"＝缺檔；"redo"＝檔案在，但版本和目前設定不同（依 STATE["redo"] 決定要不要列入）。
    sig＝聲音（Part 7 多人聊天為各段聲音用 | 相連）。"""
    if not ok_file(path):
        return "new"
    if STATE["redo"] <= 0:
        return None
    rec = STATE["log"].get(rel_key(path))
    if rec is None:
        return "redo" if STATE["redo"] >= 2 else None
    same = rec.get("engine") == ENGINE and rec.get("voice") == sig and (rec.get("rate") or "") == (rate or "")
    return None if same else "redo"


def record(path, sig, rate):
    STATE["log"][rel_key(path)] = dict(voice=sig, rate=rate or "", engine=ENGINE,
                                       date=datetime.now().strftime("%Y-%m-%d %H:%M"))
    STATE["dirty"] += 1


def save_log(root=None):
    root = Path(root or STATE["root"])
    tmp = root / (LOG_NAME + ".tmp")
    data = dict(v=1, files=dict(sorted(STATE["log"].items())))
    tmp.write_text(json.dumps(data, ensure_ascii=False, indent=1), encoding="utf-8")
    tmp.replace(log_path(root))
    STATE["dirty"] = 0


def short(v):
    return v.split("-")[-1].replace("Neural", "")


def match_key(key, p):
    """前綴比對（d1 / d2-001）；以 - 開頭則比對結尾（-e / -m / -h 依難度）。"""
    key, p = key.lower(), p.lower()
    return key.endswith(p) if p.startswith("-") else key.startswith(p)


def load_items(root: Path, fname):
    data = json.loads((root / fname).read_text(encoding="utf-8-sig"))
    return data["items"] if isinstance(data, dict) else data


def gender_voice(iid, g):
    """依性別（F／M）挑女聲／男聲，同一個 id 固定同一個。"""
    pool = VOICES[1::2] if g == "M" else VOICES[0::2]
    return pool[zlib.crc32(iid.encode("utf-8")) % len(pool)]


def letters_group(audio: Path, rate):
    if not MAKE_LETTERS:
        return None
    jobs = []
    for ch in LETTERS:
        path = audio / f"{ch}.mp3"
        r = need(path, LETTER_VOICE, rate)
        if r:
            jobs.append(dict(item="letters", name=f"{ch}.mp3", text=f"{ch}.", voice=LETTER_VOICE, rate=rate,
                             path=path, redo=(r == "redo")))
    if jobs:
        return dict(key="letters", total=len(LETTERS), voice=LETTER_VOICE, jobs=jobs)
    return None


def clip(text, n=70):
    return text if len(text) <= n else text[:n] + "…"

def plan_part1(root: Path):
    """一題的 pool 句子用同一個聲音（依題目 id 固定挑一個），模擬 TOEIC 多種口音。"""
    C = CFG["part1"]

    for it in items:
        if it.get("skip") or not it.get("pool"):
            continue
        iid = it["id"]

        jobs = []

    g = letters_group(audio, C["rate"])
    if g:
        groups.append(g)
    return groups

def pick_voices_p2(iid):
    """回傳 (問句聲音, 回答聲音)：性別相反，依題目 id 固定。"""
    h = zlib.crc32(iid.encode("utf-8"))
    qi = h % len(VOICES)
    others = [i for i in range(len(VOICES)) if i % 2 != qi % 2]
    return VOICES[qi], VOICES[others[(h >> 8) % len(others)]]

def pick_voices_p3(it):
    """回傳 {說話者 id: 聲音}。依性別分女聲／男聲，同性別的多位說話者用不同聲音；依題目 id 固定。"""
    h = zlib.crc32(it["id"].encode("utf-8"))
    fem, mal = VOICES[0::2], VOICES[1::2]
    used = {"F": 0, "M": 0}
    out = {}
    for s in it.get("speakers") or []:
        g = "M" if s.get("gender") == "M" else "F"
        pool = mal if g == "M" else fem
        out[s["id"]] = pool[(h + (h >> 8 if g == "M" else 0) + used[g]) % len(pool)]
        used[g] += 1
    return out

def strip_marks(t):
    """插入題的 [1]–[4] 標記不念（與 part7.js 的 strip 相同）。"""
    return re.sub(r"\s+", " ", re.sub(r"\s*\[[1-4]\]\s*", " ", str(t))).strip()


def voice_of(gender, key, used=0):
    """依性別（F／M）與 key 固定挑一個聲音；used＝同性別已用過幾個（讓同性別的人聲音不同）。"""
    pool = VOICES[1::2] if gender == "M" else VOICES[0::2]
    h = zlib.crc32(key.encode("utf-8"))
    return pool[((h >> 8 if gender == "M" else h) + used) % len(pool)]


def punct(t):
    """片段結尾若不是標點就補句點，串在一起時才會有停頓。"""
    return t if not t or t[-1] in ".,!?:;)\"'" else t + "."


def doc_parts(d, key):
    """回傳 ([(文字, 聲音), ...], 聲音說明)。chat 每則訊息一段，其餘整份一段。"""
    g = "M" if d.get("voice") == "M" else "F"
    if d.get("say") and d.get("kind") != "chat":
        v = voice_of(g, key)
        return [(strip_marks(d["say"]), v)], f"{g} {short(v)}"
    if d.get("kind") == "chat":
        vmap, used, desc = {}, {"F": 0, "M": 0}, []
        for who, gen in (d.get("speakers") or {}).items():
            gg = "M" if gen == "M" else "F"
            vmap[who] = voice_of(gg, key, used[gg])
            used[gg] += 1
            desc.append(f"{who.split()[0]} {short(vmap[who])}")
        dv = voice_of(g, key)
        parts = [(m["t"].strip(), vmap.get(m.get("who"), dv)) for m in d.get("messages") or [] if m.get("t")]
        return parts, " / ".join(desc)
    v = voice_of(g, key)
    lines = [punct(strip_marks(t)) for t in (d.get("head") or []) + (d.get("body") or []) if isinstance(t, str)]
    return [(" ".join(x for x in lines if x), v)], f"{g} {short(v)}"

def norm(s):
    return re.sub(r"\s+", " ", str(s or "").strip().lower())


def fnv6(s):
    h = 0x811C9DC5
    for b in s.encode("utf-8"):
        h ^= b
        h = (h * 0x01000193) & 0xFFFFFFFF
    return f"{h:08x}"[:6]


def slug(text):
    """檔名規則：必須與網頁 notes.js 的 dnSlug 完全一致，不要改。"""
    k = norm(text)
    s = re.sub(r"[^a-z0-9]+", "-", k).strip("-")
    if s and len(s) <= 40:
        return s
    return s[:40].rstrip("-") + ("-" if s else "") + fnv6(k)


# 切句（與 speak.js 的 splitSent 相同）
ABBR = re.compile(r"\b(Mr|Mrs|Ms|Dr|Prof|Inc|Ltd|Co|Corp|St|No|vs|etc)\.$")
SENT_END = re.compile(r"[.!?]+[\"')\]]*\s+(?=[\"'(\[]?[A-Z0-9])")


def split_sent(p):
    out = []
    for m in re.finditer(r"[^\n]+", p):
        line, base = m.group(0), m.start()

        def push(s, e):
            while s < e and line[s].isspace():
                s += 1
            while e > s and line[e - 1].isspace():
                e -= 1
            if e > s:
                out.append((base + s, base + e))

        st = 0
        for k in SENT_END.finditer(line):
            if ABBR.search(line[st:k.start() + 1]):
                continue
            push(st, k.end())
            st = k.end()
        push(st, len(line))
    return out


def sentence_of(passage, text):
    i = passage.lower().find(str(text).lower())
    if i < 0:
        return None
    for a, b in split_sent(passage):
        if a <= i < b:
            return passage[a:b].strip()
    return None


def speak_text(t):
    return re.sub(r"\.{2,}|…", " ", str(t)).strip()

def collect_entries(root, items):
    """詳解項目：extra_vocab ＋ daily-marks*.json。回傳 list of dict(qid, text, type, ctx, examples)"""
    arts = {f"w{x['week']}d{x['day']}": x for x in items}
    ents = {}
    for qid, x in arts.items():
        for v in x.get("extra_vocab") or []:
            if not isinstance(v, dict):
                continue
            t = str(v.get("text") or v.get("word") or "").strip()
            if t:
                typ = v.get("type") or ("phrase" if re.search(r"\s", t) else "word")
                ents[(qid, norm(t))] = dict(qid=qid, text=t, type=typ, ctx=None, examples=v.get("examples") or [],
                                            collocations=v.get("collocations") or [], forms=v.get("forms") or [],
                                            family=v.get("family") or [], derived=v.get("derived") or [],
                                            confusable=v.get("confusable") or [], similar=v.get("similar") or [],
                                            en=str(v.get("en") or "").strip(), key_words=v.get("key_words") or [], paraphrase=v.get("paraphrase") or [],
                                            pos_uses=v.get("pos_uses") or [], senses=v.get("senses") or [],
                                            lemma=str(v.get("lemma") or "").strip())
    for f in sorted(root.glob("daily-marks*.json")):
        try:
            j = json.loads(f.read_text(encoding="utf-8-sig"))
        except Exception:
            continue
        for m in (j if isinstance(j, list) else (j.get("marks") or [])):
            if not isinstance(m, dict) or m.get("qid") not in arts or not str(m.get("text") or "").strip():
                continue
            t = str(m["text"]).strip()
            k = (m["qid"], norm(t))
            e = ents.setdefault(k, dict(qid=m["qid"], text=t, type=m.get("type") or "word", ctx=None, examples=[],
                                        collocations=[], forms=[], family=[], derived=[], confusable=[], pos_uses=[], senses=[], lemma="", en="", key_words=[], paraphrase=[]))
            if m.get("ctx"):
                e["ctx"] = str(m["ctx"]).strip()
    out = []
    for e in ents.values():
        if e["type"] != "sentence" and not e["ctx"]:
            e["ctx"] = sentence_of(str(arts[e["qid"]].get("passage") or ""), e["text"])
        out.append(e)
    out.sort(key=lambda e: (e["qid"], norm(e["text"])))
    return out


def plan_daily(root: Path):
    """回傳 groups：list of dict(key, kind, total, voice, jobs[])，只包含有缺檔的項目。"""
    begin(root)
    items = load_daily_items(root)
    audio = root / "audio"
    groups = []
    if MAKE_PASSAGE:
        for x in items:
            qid = f"w{x['week']}d{x['day']}"
            p = str(x.get("passage") or "").strip()
            if not p:
                continue
            path = (root / x["audio"]) if x.get("audio") else audio / "daily" / f"{qid}.mp3"
            voice = PASSAGE_VOICES[zlib.crc32(qid.encode("utf-8")) % len(PASSAGE_VOICES)]
            r = need(path, voice, PASSAGE_RATE)
            if r:
                groups.append(dict(key=qid, kind="文章", total=1, voice=voice, jobs=[
                    dict(name=path.name, text=p, voice=voice, rate=PASSAGE_RATE, path=path, redo=(r == "redo"))]))
    if MAKE_VOCAB or MAKE_CTX or MAKE_EXAMPLES or MAKE_COLLOC or MAKE_FORMS or MAKE_FAMILY or MAKE_DERIVED or MAKE_CONFUSABLE or MAKE_SIMILAR or MAKE_EN or MAKE_PARA or MAKE_KEYWORDS or MAKE_POS or MAKE_SENSE:
        vdir = root / VOCAB_SUB
        wdir = root / WORDS_SUB
        seen = set()   # 這次已排進清單的檔（共用單字檔、vocab/ 檔），同一個檔不重複排
        legacy = [f.name for f in vdir.glob("*.mp3") if re.match(r"w\d+d\d+-", f.name)] if vdir.is_dir() else []
        if legacy:
            print(f"⚠ vocab/ 還有 {len(legacy)} 個舊檔名（{legacy[0]} …）。請先執行 migrate_vocab_names.py，否則下面會把它們當成「缺檔」重錄一次。")
        shared = lambda t: f"@{slug(t)}.mp3"   # 「@」開頭＝共用發音庫 words/ 的檔
        for e in collect_entries(root, items):
            base = slug(e["text"])   # 不含文章 id：例句等跟著單字；整句與所在句另用「句子本身」的 slug（見下）
            want = []
            if MAKE_VOCAB:
                want.append((f"{base}.mp3" if e["type"] == "sentence" else shared(e["text"]), speak_text(e["text"])))
            if MAKE_VOCAB and e.get("lemma") and norm(e["lemma"]) != norm(e["text"]):
                want.append((shared(e["lemma"]), speak_text(e["lemma"])))  # 原形的發音
            if MAKE_CTX and e["type"] != "sentence" and e["ctx"]:
                want.append((f"{slug(e['ctx'])}.mp3", e["ctx"]))   # 所在句：依句子內容命名（與入庫的整句共用同一檔）
            if MAKE_EXAMPLES:
                for k, ex in enumerate(e["examples"], 1):
                    en = ex if isinstance(ex, str) else (ex or {}).get("en")
                    if en and str(en).strip():
                        want.append((f"{base}_e{k}.mp3", str(en).strip()))
            if MAKE_COLLOC:
                for k, c in enumerate(e["collocations"], 1):
                    ph = c if isinstance(c, str) else ((c or {}).get("phrase") or (c or {}).get("p"))
                    if isinstance(ph, str) and ph.strip():
                        want.append((shared(ph), speak_text(ph)))
                    en = ((c or {}).get("example") or {}).get("en") if isinstance(c, dict) else None
                    if isinstance(en, str) and en.strip():
                        want.append((f"{base}_c{k}.mp3", en.strip()))
            if MAKE_FORMS:
                for k, f in enumerate(e["forms"], 1):
                    w = f.get("w") if isinstance(f, dict) else None
                    if isinstance(w, str) and w.strip():
                        want.append((shared(w), speak_text(w)))
                    en = (f.get("example") or {}).get("en") if isinstance(f, dict) and isinstance(f.get("example"), dict) else None
                    if isinstance(en, str) and en.strip():   # 屈折變化的例句：檔名 _fx{k}，與 notes.js 詳解頁的 🔊 一致
                        want.append((f"{base}_fx{k}.mp3", en.strip()))
            if MAKE_FAMILY:
                for k, g in enumerate(e["family"], 1):
                    if not isinstance(g, dict):
                        continue
                    if isinstance(g.get("w"), str) and g["w"].strip():
                        want.append((shared(g["w"]), speak_text(g["w"])))
                    en = (g.get("example") or {}).get("en") if isinstance(g.get("example"), dict) else None
                    if isinstance(en, str) and en.strip():
                        want.append((f"{base}_mx{k}.mp3", en.strip()))
            if MAKE_DERIVED:
                for k, d in enumerate(e["derived"], 1):
                    w = d.get("w") if isinstance(d, dict) else None
                    if isinstance(w, str) and w.strip():
                        want.append((shared(w), speak_text(w)))
            if MAKE_CONFUSABLE:
                for k, c in enumerate(e.get("confusable") or [], 1):  # 易混淆：該字／片語本身發音（共用庫）＋ 例句 _cf{k}，與 notes.js 的 🔊 一致
                    cw = (c.get("word") or c.get("phrase")) if isinstance(c, dict) else None
                    if isinstance(cw, str) and cw.strip():
                        want.append((shared(cw), speak_text(cw)))
                    en = (c.get("example") or {}).get("en") if isinstance(c, dict) and isinstance(c.get("example"), dict) else None
                    if isinstance(en, str) and en.strip():
                        want.append((f"{base}_cf{k}.mp3", en.strip()))
            if MAKE_EN and e.get("en"):   # 英文解說整句：逐篇存放，檔名 _en，與 notes.js 的 🔊 一致
                want.append((f"{base}_en.mp3", str(e["en"]).strip()))
            if MAKE_PARA:                 # 句子的同義改寫：逐篇存放，檔名 _pr{k}，與 notes.js 的 🔊 一致
                for k, o in enumerate(e.get("paraphrase") or [], 1):
                    en = o.get("en") if isinstance(o, dict) else (o if isinstance(o, str) else None)
                    if isinstance(en, str) and en.strip():
                        want.append((f"{base}_pr{k}.mp3", en.strip()))
            if MAKE_KEYWORDS:             # 句中值得注意的字：單字走共用庫
                for k, o in enumerate(e.get("key_words") or [], 1):
                    kw = o.get("w") if isinstance(o, dict) else None
                    if isinstance(kw, str) and kw.strip():
                        want.append((shared(kw), speak_text(kw)))
            if MAKE_SIMILAR:
                for k, c in enumerate(e.get("similar") or [], 1):  # 近義片語：本身發音（共用庫）＋ 例句 _sx{k}，與 notes.js 的 🔊 一致
                    if not isinstance(c, dict):
                        continue
                    sw = c.get("phrase") or c.get("word")
                    if isinstance(sw, str) and sw.strip():
                        want.append((shared(sw), speak_text(sw)))
                    en = (c.get("example") or {}).get("en") if isinstance(c.get("example"), dict) else None
                    if isinstance(en, str) and en.strip():
                        want.append((f"{base}_sx{k}.mp3", en.strip()))
            for on, key, wsuf, esuf in ((MAKE_POS, "pos_uses", "pw", "p"), (MAKE_SENSE, "senses", "sw", "s")):
                if not on:
                    continue
                for k, o in enumerate(e[key], 1):  # 每一筆：單字發音（_pw／_sw）＋ 例句（_p／_s）；網頁 notes.js 的 🔊 用同一套檔名
                    if not isinstance(o, dict):
                        continue
                    ww = o.get("w") or e.get("lemma") or e["text"]
                    want.append((shared(ww), speak_text(ww)))  # 變化形（billed）的詳解以原形（bill）為準
                    en = (o.get("example") or {}).get("en") if isinstance(o.get("example"), dict) else None
                    if isinstance(en, str) and en.strip():
                        want.append((f"{base}_{esuf}{k}.mp3", en.strip()))
            jobs = []
            for n, t in want:
                if n.startswith("@"):   # 單字類：共用庫；已存在、或這次已排進別的項目就不重複
                    p = wdir / n[1:]
                    r = need(p, VOCAB_VOICE, VOCAB_RATE)
                    if not r or p in seen:
                        continue
                    seen.add(p)
                else:                   # 例句、所在句、句子：放 vocab/；同一句（所在句／整句／兩個字同一句）只排一次
                    p = vdir / n
                    r = need(p, VOCAB_VOICE, VOCAB_RATE)
                    if not r or p in seen:
                        continue
                    seen.add(p)
                jobs.append(dict(name=p.name, text=t, voice=VOCAB_VOICE, rate=VOCAB_RATE, path=p, redo=(r == "redo")))
            if jobs:
                kind = {"word": "單字", "phrase": "片語", "sentence": "句子"}.get(e["type"], "單字")
                groups.append(dict(key=f"{e['qid']}:{e['text']}", kind=kind, total=len(want), voice=VOCAB_VOICE, jobs=jobs))
    return groups


def write_manifest(root: Path, log=print):

class Spec:
    def __init__(self, key, title, plan, unit="題", id_hdr="題目", sample_hdr="第一句", voice_w=160, sample_w=520,
                 letters=False, log_style="plain", detail_sp=False, cli_vw=20, prefix_hint="d1 / d2-001 / -e 初級 / -m 中級 / -h 高級",
                 cli_prefix="前綴：d1 或 d2-001　難度：-e / -m / -h", daily=False):
        self.key, self.title, self.plan, self.unit = key, title, plan, unit
        self.id_hdr, self.sample_hdr, self.voice_w, self.sample_w = id_hdr, sample_hdr, voice_w, sample_w
        self.letters, self.log_style, self.detail_sp = letters, log_style, detail_sp
        self.cli_vw, self.prefix_hint, self.cli_prefix, self.daily = cli_vw, prefix_hint, cli_prefix, daily
        self.cfg = CFG[key]

    @property
    def file(self):
        return self.cfg["file"]

    @property
    def sub(self):
        return self.cfg["sub"]

    def est_minutes(self, n_files):
        return max(1, round(n_files * self.cfg["secs"] / CONCURRENCY / 60))

    def log_text(self, job):
        """產生成功時的記錄行文字。"""
        if self.log_style == "sp":
            return f"{job['sp']}: {job['text']}"
        if self.log_style == "cut60…":
            return job["text"][:60] + "…"
        if self.log_style == "cut60":
            return job["text"][:60]
        return job["text"]

    def folder_note(self, root):
        if self.daily:
            return f"專案：{root}　　文章：{root / 'audio' / 'daily'}　　詳解：{root / VOCAB_SUB}"
        if self.letters:
            return f"專案：{root}　　句子存放：{root / 'audio' / self.sub}　　字母存放：{root / 'audio'}"
        return f"專案：{root}　　句子存放：{root / 'audio' / self.sub}"

def have_edge_tts():
    try:
        importlib.invalidate_caches()
        importlib.import_module("edge_tts")
        return True
    except ImportError:

async def synth_edge(job, spec, sem, st, log, edge_tts):
    async with sem:
        if st["stop"].is_set():
            st["skipped"] += 1
            return
        job["path"].parent.mkdir(parents=True, exist_ok=True)
        tmp = job["path"].with_suffix(".tmp")
        parts = job.get("parts") or [(job["text"], job["voice"])]
        rate = job.get("rate") or spec.cfg["rate"]
        sig = "|".join(v for _, v in parts)
        timeout = spec.cfg["timeout"]
        err = None
        for _ in range(3):
            tmps = []
            try:
                if len(parts) == 1:
                    await asyncio.wait_for(edge_tts.Communicate(parts[0][0], parts[0][1], rate=rate).save(str(tmp)), timeout)
                else:
                    for n, (text, voice) in enumerate(parts):
                        pt = job["path"].with_suffix(f".p{n}.tmp")
                        tmps.append(pt)
                        await asyncio.wait_for(edge_tts.Communicate(text, voice, rate=rate).save(str(pt)), timeout)
                        if not (pt.exists() and pt.stat().st_size > 1000):
                            raise RuntimeError("產生的檔案太小")
                    with open(tmp, "wb") as out:
                        for pt in tmps:
                            out.write(pt.read_bytes())
                if tmp.exists() and tmp.stat().st_size > 1000:
                    tmp.replace(job["path"])
                    st["ok"] += 1
                    record(job["path"], sig, rate)
                    if STATE["dirty"] >= 20:
                        save_log()
                    log(f"{'↻' if job.get('redo') else '✓'} {job['name']}  {spec.log_text(job)}")
                    return
            except Exception as e:
                err = f"{type(e).__name__} {e}".strip()
            finally:
                for pt in tmps:
                    try:
                        pt.unlink()
                    except OSError:
                        pass
            await asyncio.sleep(1.5)
        st["bad"] += 1
        log(f"✗ 失敗 {job['name']}  ({err})")
        try:
            tmp.unlink()
        except OSError:
            pass


# ------------------------------- 語音產生（Azure Speech REST + SSML） -------------------------------
class AzureFatal(Exception):
    """金鑰／區域／聲音設定錯誤：重試沒有用，整批停止。"""


def azure_conf():
    """從環境變數讀金鑰與區域。回傳 (key, region, 錯誤訊息或 None)。不會把金鑰印出來。"""
    key = os.environ.get("SPEECH_KEY", "").strip()
    region = os.environ.get("SPEECH_REGION", "").strip().lower().replace(" ", "")
    miss = [n for n, v in (("SPEECH_KEY", key), ("SPEECH_REGION", region)) if not v]
    if miss:
        return key, region, ("找不到環境變數 " + "、".join(miss) + "。請在命令提示字元執行：setx SPEECH_KEY \"金鑰\" 和 "
                             "setx SPEECH_REGION \"區域\"，然後關掉所有終端機／編輯器重開（雙擊執行則請登出再登入）。")
    if not re.fullmatch(r"[a-z0-9]+", region):
        return key, region, "SPEECH_REGION 格式不對，應為小寫英數字（例如 eastasia、japaneast），不是「East Asia」。"
    return key, region, None


def tts_url(region):
    return f"https://{region}.tts.speech.microsoft.com/cognitiveservices/v1"


def prosody_open(rate):
    r = (rate or "").strip()
    return (f'<prosody rate="{xml_escape(r)}">', "</prosody>") if r and r not in ("+0%", "0%") else ("", "")


def clean_text(t):
    t = re.sub(r"[\x00-\x08\x0b\x0c\x0e-\x1f]", "", t)   # XML 不允許的控制字元
    return xml_escape(re.sub(r"\s+", " ", t).strip())


def build_ssml(parts, rate):
    """parts＝[(文字, 聲音), ...]。多段用同一份 SSML 的多個 <voice>，一次請求就得到接好的 mp3。"""
    po, pc = prosody_open(rate)
    brk = f'<break time="{PAUSE_MS}ms"/>' if len(parts) > 1 else ""
    body = []
    for i, (text, voice) in enumerate(parts):
        t = clean_text(text)
        if not t:
            raise ValueError("文字是空的")
        body.append(f'<voice name="{xml_escape(voice)}">{po}{t}{pc}{brk if i < len(parts) - 1 else ""}</voice>')
    return "<speak version='1.0' xml:lang='en-US'>" + "".join(body) + "</speak>"


def job_chars(job):
    parts = job.get("parts") or [(job["text"], job["voice"])]
    return sum(len(clean_text(t)) for t, _ in parts)


def azure_request(key, region, ssml, timeout):
    """同步呼叫一次；成功回傳 mp3 bytes；失敗丟 urllib.error.HTTPError／URLError。"""
    req = urllib.request.Request(
        tts_url(region), data=ssml.encode("utf-8"), method="POST",
        headers={"Ocp-Apim-Subscription-Key": key, "Content-Type": "application/ssml+xml",
                 "X-Microsoft-OutputFormat": AZURE_FORMAT, "User-Agent": "toeic-auto-speech"})
    with urllib.request.urlopen(req, timeout=timeout or 60) as r:
        return r.read()


async def azure_synth(parts, rate, timeout, conf):
    """含重試／限流（429 依 Retry-After 等待、5xx 與網路錯誤指數退避）。401／403／404／400 不重試。"""
    key, region = conf
    ssml = build_ssml(parts, rate)
    last = None
    for attempt in range(AZURE_RETRIES):
        try:
            return await asyncio.to_thread(azure_request, key, region, ssml, timeout)
        except urllib.error.HTTPError as e:
            code = e.code
            detail = ""
            try:
                detail = e.read().decode("utf-8", "replace")[:120]
            except Exception:
                pass
            if code in (401, 403):
                raise AzureFatal(f"HTTP {code}：金鑰無效、區域和金鑰不符，或此訂用帳戶已停用／額度用完。{detail}")
            if code == 404:
                raise AzureFatal("HTTP 404：端點不存在，請確認 SPEECH_REGION 和資源所在區域一致。")
            if code in (400, 415):
                raise ValueError(f"HTTP {code}：請求被拒絕（SSML 或聲音名稱有問題）{detail}")
            if code == 429 or code >= 500:
                wait = 0.0
                try:
                    wait = float(e.headers.get("Retry-After") or 0)
                except (TypeError, ValueError):
                    pass
                wait = max(wait, min(2 ** attempt, 30)) + random.uniform(0, 1)
                last = f"HTTP {code}"
                await asyncio.sleep(wait)
                continue
            raise RuntimeError(f"HTTP {code} {detail}")
        except (urllib.error.URLError, TimeoutError, ConnectionError, OSError) as e:
            last = f"{type(e).__name__} {e}".strip()
            await asyncio.sleep(min(2 ** attempt, 30) + random.uniform(0, 1))
    raise RuntimeError(f"重試 {AZURE_RETRIES} 次仍失敗（{last}）")


async def synth(job, spec, sem, st, log, conf):
    async with sem:
        if st["stop"].is_set():
            st["skipped"] += 1
            return
        job["path"].parent.mkdir(parents=True, exist_ok=True)
        tmp = job["path"].with_suffix(".tmp")
        parts = job.get("parts") or [(job["text"], job["voice"])]   # [(文字, 聲音), ...]；多段在同一份 SSML 內依序朗讀
        rate = job.get("rate") or spec.cfg["rate"]
        sig = "|".join(v for _, v in parts)             # 版本紀錄用的聲音簽名（和 plan 裡的 need() 一致）
        timeout = spec.cfg["timeout"]
        try:
            data = await azure_synth(parts, rate, timeout, conf)
            if len(data) <= 1000:
                raise RuntimeError("產生的檔案太小")
            tmp.write_bytes(data)
            tmp.replace(job["path"])
            st["ok"] += 1
            USAGE["session"] += job_chars(job)
            record(job["path"], sig, rate)
            if STATE["dirty"] >= 20:                     # 每 20 筆存一次，中途當掉也不會全部遺失
                save_log()
            log(f"{'↻' if job.get('redo') else '✓'} {job['name']}  {spec.log_text(job)}")
        except AzureFatal as e:
            st["bad"] += 1
            log(f"✗ 失敗 {job['name']}  ({e})")
            if not st.get("fatal"):
                st["fatal"] = True
                log("✗ 這類錯誤重試沒有用，已停止其餘檔案。請修正設定後再執行（可先跑 --check-azure）。")
                st["stop"].set()
        except Exception as e:
            st["bad"] += 1
            log(f"✗ 失敗 {job['name']}  ({type(e).__name__} {e})".strip())
        finally:
            try:
                tmp.unlink()
            except OSError:
                pass


def generate(jobs, spec, log, progress, stop, root=None):
    """同步執行（給文字模式或背景執行緒呼叫）。回傳 (成功, 失敗, 略過)"""
    if root is not None:
        STATE["root"] = Path(root)
    st = dict(ok=0, bad=0, skipped=0, stop=stop)
    total = len(jobs)

    def lg(m):
        log(m)
        progress(st["ok"] + st["bad"], total)

    if ENGINE == "azure":
        key, region, err = azure_conf()
        if err:
            log("✗ " + err)
            return 0, len(jobs), 0
        chars = sum(job_chars(j) for j in jobs)
        USAGE["session"] = 0
        log(f"引擎：Azure Speech　區域 {region}；本次 {total} 個檔、約 {chars:,} 字元"
            f"（估算約 US${chars / 1e6 * PRICE_USD_PER_M_CHARS:.3f}）")
        log(credit_line(STATE["root"]))

        async def go():
            sem = asyncio.Semaphore(CONCURRENCY)
            await asyncio.gather(*(synth(j, spec, sem, st, lg, (key, region)) for j in jobs))
    else:
        if not have_edge_tts():
            log("正在安裝 edge-tts …")
            r = subprocess.run([sys.executable, "-m", "pip", "install", "edge-tts"], capture_output=True, text=True)
            if r.returncode != 0 or not have_edge_tts():
                log("✗ 安裝 edge-tts 失敗，請手動執行：pip install edge-tts")
                return 0, len(jobs), 0
        import edge_tts
        log("引擎：edge-tts（免費）。注意：2026-10-10 起微軟改版，常見 HTTP 403 失敗，與本程式無關。")

        async def go():
            sem = asyncio.Semaphore(CONCURRENCY)
            await asyncio.gather(*(synth_edge(j, spec, sem, st, lg, edge_tts) for j in jobs))

    try:
        asyncio.run(go())
    finally:
        if STATE["dirty"] and STATE["root"]:
            try:
                save_log()
            except OSError as e:
                log(f"✗ 無法寫入版本紀錄 {LOG_NAME}：{e}")
        if ENGINE == "azure" and STATE["root"]:
            try:
                save_usage(STATE["root"])
                log(credit_line(STATE["root"]))
            except OSError as e:
                log(f"✗ 無法寫入用量紀錄 {USAGE_NAME}：{e}")
    return st["ok"], st["bad"], st["skipped"]


def check_azure():
    """--check-azure：驗證金鑰／區域，列出 8 個聲音是否存在，並實際合成一次很短的句子。"""
    key, region, err = azure_conf()
    if err:
        print("✗", err)
        return 1
    print(f"區域：{region}　金鑰：已讀到（長度 {len(key)}，不顯示內容）")
    try:
        req = urllib.request.Request(f"https://{region}.tts.speech.microsoft.com/cognitiveservices/voices/list",
                                     headers={"Ocp-Apim-Subscription-Key": key})
        with urllib.request.urlopen(req, timeout=30) as r:
            have = {v["ShortName"] for v in json.loads(r.read().decode("utf-8"))}
    except urllib.error.HTTPError as e:
        print(f"✗ 讀取聲音清單失敗：HTTP {e.code}（401／403＝金鑰或區域不對；404＝區域名稱錯）")
        return 1
    except Exception as e:
        print(f"✗ 連線失敗：{type(e).__name__} {e}")
        return 1
    bad = 0
    for v in dict.fromkeys(VOICES + [LETTER_VOICE, VOCAB_VOICE]):
        ok = v in have
        bad += (not ok)
        print(f"  {'✓' if ok else '✗'} {v}")
    try:
        data = azure_request(key, region, build_ssml([("Hello.", VOICES[0])], "-5%"), 30)
        print(f"✓ 合成測試成功（{len(data):,} bytes，約 6 個字元）")
    except Exception as e:
        print(f"✗ 合成測試失敗：{type(e).__name__} {e}")
        return 1
    print(credit_line(find_root() or HERE))
    print("全部可用。" if not bad else f"有 {bad} 個聲音在 {region} 不可用，請改用別的聲音或區域。")
    return 1 if bad else 0


def run_scan(root: Path, log):
    scan = root / "Scan總表與音檔.py"
    if not scan.exists():
        log("（找不到 Scan總表與音檔.py，請自行更新 index.json）")
        return
    r = subprocess.run([sys.executable, str(scan)], cwd=str(scan.parent), capture_output=True, text=True, input="\n")
    log((r.stdout or "") + (r.stderr or "") or "Scan總表與音檔.py 已執行。")

def run_gui(root: Path, spec: Spec):
    """回傳 "switch"（按了「切換題庫」）或 None。"""
    import tkinter as tk
    from tkinter import ttk, messagebox

    win = tk.Tk()
    win.title(f"{spec.title} 補錄音檔")
    win.geometry("1000x720")
    S = dict(groups=[], checked=set(), running=False, stop=threading.Event(), result=None)
    q = queue.Queue()
    unit = spec.unit
    kind_btns = {}   # Daily 的「文章／詳解」切換鈕：{True: 文章鈕, False: 詳解鈕}

    ttk.Label(win, text=spec.folder_note(root), foreground="#555").pack(anchor="w", padx=10, pady=(8, 2))

    bar = ttk.Frame(win)
    bar.pack(fill="x", padx=10, pady=4)
    bar2 = ttk.Frame(win)
    bar2.pack(fill="x", padx=10, pady=(0, 4))
    n_var, pre_var = tk.StringVar(value="10"), tk.StringVar()

    tree_f = ttk.Frame(win)
    tree_f.pack(fill="both", expand=True, padx=10)
    if spec.daily:
        cols = ("chk", "key", "kind", "miss", "sample")
        heads, widths = ("勾", spec.id_hdr, "類型", "缺/總", "內容"), (40, 200, 60, 90, 560)
        centered = ("chk", "kind", "miss")
    else:
        cols = ("chk", "id", "miss", "voice", "sample")
        heads, widths = ("勾", spec.id_hdr, "缺/總", "聲音", spec.sample_hdr), (40, 110, 90, spec.voice_w, spec.sample_w)
        centered = ("chk", "miss")
    tree = ttk.Treeview(tree_f, columns=cols, show="headings", selectmode="browse", height=12)
    for c, t, w in zip(cols, heads, widths):
        tree.heading(c, text=t)
        tree.column(c, width=w, anchor="center" if c in centered else "w", stretch=(c == "sample"))
    sb = ttk.Scrollbar(tree_f, orient="vertical", command=tree.yview)
    tree.configure(yscrollcommand=sb.set)
    tree.pack(side="left", fill="both", expand=True)
    sb.pack(side="right", fill="y")

    summary = ttk.Label(win, font=("", 11, "bold"))
    summary.pack(anchor="w", padx=10, pady=(6, 0))
    detail = tk.Text(win, height=7, wrap="none", background="#f7f7f7")
    detail.pack(fill="x", padx=10, pady=4)

    act = ttk.Frame(win)
    act.pack(fill="x", padx=10)
    prog = ttk.Progressbar(win, mode="determinate")
    prog.pack(fill="x", padx=10, pady=4)
    logbox = tk.Text(win, height=8, wrap="none", background="#101418", foreground="#d6e2ee")
    logbox.pack(fill="both", padx=10, pady=(0, 10))

    def log(m):
        logbox.insert("end", m + "\n")
        logbox.see("end")

    def sample_of(g):
        if spec.daily:
            return g["jobs"][0]["text"][:80].replace("\n", " ")
        return g.get("sample") or g["jobs"][0]["text"]

    def update_summary():
        g = [S["groups"][int(i)] for i in S["checked"]]
        n = sum(len(x["jobs"]) for x in g)
        total = sum(len(x["jobs"]) for x in S["groups"])
        rn = sum(1 for x in g for j in x["jobs"] if j.get("redo"))
        rt = sum(1 for x in S["groups"] for j in x["jobs"] if j.get("redo"))
        summary.config(text=f"已勾選 {len(g)} {unit}，共 {n} 個檔" + (f"（其中 {rn} 個重錄）" if rn else "")
                       + f"，預估約 {spec.est_minutes(n) if n else 0} 分鐘　（全部 {total} 個檔" + (f"，含重錄 {rt} 個" if rt else "") + "）")
        start_btn.config(state="normal" if (n and not S["running"]) else "disabled")
        refresh_kind_btns()

    def row_values(i, g):
        chk = "☑" if str(i) in S["checked"] else "☐"
        rn = sum(1 for j in g["jobs"] if j.get("redo"))
        miss = f"{len(g['jobs'])}/{g['total']}" + (f" ↻{rn}" if rn else "")
        if spec.daily:
            return (chk, g["key"], g["kind"], miss, sample_of(g))
        return (chk, g["key"], miss, g["voice"], sample_of(g))

    def repaint():
        tree.delete(*tree.get_children())
        for i, g in enumerate(S["groups"]):
            tree.insert("", "end", iid=str(i), values=row_values(i, g))
        update_summary()

    def reload_plan():
        S["groups"] = spec.plan(root)
        S["checked"] = set()
        repaint()
        detail.delete("1.0", "end")

    def set_checked(ids):
        S["checked"] = set(map(str, ids))
        repaint()

    def toggle(iid):
        S["checked"].symmetric_difference_update({iid})
        tree.set(iid, "chk", "☑" if iid in S["checked"] else "☐")
        update_summary()

    def on_click(e):
        if tree.identify_region(e.x, e.y) == "cell" and tree.identify_column(e.x) == "#1":
            iid = tree.identify_row(e.y)
            if iid:
                toggle(iid)
                return "break"

    def on_select(_e=None):
        sel = tree.selection()
        if not sel:
            return
        g = S["groups"][int(sel[0])]
        detail.delete("1.0", "end")
        if spec.daily:
            detail.insert("end", f"{g['key']}　聲音 {g['voice']}\n")
            for j in g["jobs"]:
                detail.insert("end", f" {'↻' if j.get('redo') else ' '}{j['name']:<34} {j['text'][:70].replace(chr(10), ' ')}\n")
            return
        folder = (root / "audio") if g["key"] == "letters" else (root / "audio" / spec.sub)
        detail.insert("end", f"{g['key']}　聲音 {g['voice']}　存到 {folder}\n")
        for j in g["jobs"]:
            body = f"{j['sp']}: {j['text']}" if (spec.detail_sp and "sp" in j) else j["text"]
            detail.insert("end", f" {'↻' if j.get('redo') else ' '}{j['name']:<22} {body}\n")

    def space(_e):
        sel = tree.selection()
        if sel:
            toggle(sel[0])

    tree.bind("<Button-1>", on_click)
    tree.bind("<<TreeviewSelect>>", on_select)
    tree.bind("<space>", space)

    def first_n():
        try:
            n = max(0, int(n_var.get()))
        except ValueError:
            return
        set_checked(range(min(n, len(S["groups"]))))

    def by_prefix():
        p = pre_var.get().strip().lower()
        if p:
            set_checked(i for i, g in enumerate(S["groups"]) if match_key(g["key"], p))

    def kind_ids(passage):
        return {str(i) for i, g in enumerate(S["groups"]) if (g["kind"] == "文章") == passage}

    def toggle_kind(passage):
        """該類（文章／詳解）已經全勾 → 全部取消；否則 → 全部勾選。不影響另一類的勾選狀態。"""
        ids = kind_ids(passage)
        if not ids:
            return
        if ids <= S["checked"]:
            S["checked"] -= ids
        else:
            S["checked"] |= ids
        repaint()

    def refresh_kind_btns():
        for passage, btn in kind_btns.items():
            ids = kind_ids(passage)
            name = "文章" if passage else "詳解"
            on = bool(ids) and ids <= S["checked"]
            btn.config(text=(f"☑ 取消全部{name}" if on else f"☐ 勾選全部{name}"),
                       state="normal" if ids else "disabled")

    ttk.Button(bar, text="全選", command=lambda: set_checked(range(len(S["groups"])))).pack(side="left")
    ttk.Button(bar, text="全不選", command=lambda: set_checked([])).pack(side="left", padx=4)
    if spec.daily:
        kind_btns[True] = ttk.Button(bar, text="☐ 勾選全部文章", command=lambda: toggle_kind(True))
        kind_btns[True].pack(side="left")
        kind_btns[False] = ttk.Button(bar, text="☐ 勾選全部詳解", command=lambda: toggle_kind(False))
        kind_btns[False].pack(side="left", padx=4)
    else:
        ttk.Button(bar, text="反選", command=lambda: set_checked(i for i in range(len(S["groups"])) if str(i) not in S["checked"])).pack(side="left")
    ttk.Separator(bar, orient="vertical").pack(side="left", fill="y", padx=10)
    ttk.Label(bar, text="只勾前").pack(side="left")
    ttk.Entry(bar, textvariable=n_var, width=4).pack(side="left", padx=2)
    if spec.daily:
        ttk.Button(bar, text=unit, command=first_n).pack(side="left", padx=2)
    else:
        ttk.Label(bar, text=unit).pack(side="left")
        ttk.Button(bar, text="勾選", command=first_n).pack(side="left", padx=4)
    ttk.Separator(bar, orient="vertical").pack(side="left", fill="y", padx=10)
    ttk.Label(bar, text=f"依前綴勾選（{spec.prefix_hint}）" if not spec.daily else f"依前綴（{spec.prefix_hint}）").pack(side="left")
    ttk.Entry(bar, textvariable=pre_var, width=10).pack(side="left", padx=4)
    ttk.Button(bar, text="勾選", command=by_prefix).pack(side="left")

    def start():
        jobs = [j for i in sorted(S["checked"], key=int) for j in S["groups"][int(i)]["jobs"]]
        if not jobs:
            return
        S["running"] = True
        S["stop"].clear()
        start_btn.config(state="disabled")
        stop_btn.config(state="normal")
        switch_btn.config(state="disabled")
        redo_cb.config(state="disabled")
        eng_cb.config(state="disabled")
        prog.config(maximum=len(jobs), value=0)
        log(f"—— 開始產生 {len(jobs)} 個檔案（需要網路）——")

        def work():
            try:
                res = generate(jobs, spec, lambda m: q.put(("log", m)), lambda d, t: q.put(("prog", d)), S["stop"], root)
            except Exception as e:
                q.put(("log", f"✗ 發生錯誤：{e}"))
                res = (0, len(jobs), 0)
            q.put(("done", res))

        threading.Thread(target=work, daemon=True).start()

    def finish(res):
        ok, bad, skipped = res
        S["running"] = False
        stop_btn.config(state="disabled")
        switch_btn.config(state="normal")
        redo_cb.config(state="readonly")
        eng_cb.config(state="readonly")
        log(f"—— 完成：成功 {ok}，失敗 {bad}" + (f"，已取消 {skipped}" if skipped else "") + " ——")
        if spec.daily:
            write_manifest(root, log)
        reload_plan()
        if ok and (root / "Scan總表與音檔.py").exists():
            tip = ("要順便執行 Scan總表與音檔.py 更新 audio/index.json 嗎？\n（文章整篇 mp3 有新增時才需要；網頁之後按 Ctrl+F5 重新整理）"
                   if spec.daily else "要順便執行 Scan總表與音檔.py 更新 index.json 嗎？\n（網頁之後按 Ctrl+F5 重新整理）")
            if messagebox.askyesno("更新 index.json", tip):
                run_scan(root, log)

    def poll():
        try:
            while True:
                k, v = q.get_nowait()
                if k == "log":
                    log(v)
                elif k == "prog":
                    prog.config(value=v)
                elif k == "done":
                    finish(v)
        except queue.Empty:
            pass
        S["poll_id"] = win.after(100, poll)

    def go_switch():
        if S["running"]:
            return
        S["result"] = "switch"
        if S.get("poll_id"):
            win.after_cancel(S["poll_id"])
        win.destroy()

    start_btn = ttk.Button(act, text="▶ 開始產生勾選的", command=start, state="disabled")
    start_btn.pack(side="left")
    stop_btn = ttk.Button(act, text="■ 停止（做完手上的就停）", command=lambda: (S["stop"].set(), log("要求停止…")), state="disabled")
    stop_btn.pack(side="left", padx=6)
    ttk.Button(act, text="重新掃描", command=lambda: None if S["running"] else reload_plan()).pack(side="left")
    switch_btn = ttk.Button(act, text="切換題庫", command=go_switch)
    switch_btn.pack(side="left", padx=6)
    ttk.Button(act, text="執行 Scan總表與音檔.py", command=lambda: run_scan(root, log)).pack(side="right")
    if spec.daily:
        ttk.Button(act, text="重建 vocab/index.json", command=lambda: write_manifest(root, log)).pack(side="right", padx=4)

    REDO_CHOICES = ["只補缺的檔", "補缺 ＋ 聲音／語速和目前設定不同的檔", "補缺 ＋ 不同的檔 ＋ 沒有紀錄的舊檔"]
    redo_var = tk.StringVar(value=REDO_CHOICES[STATE["redo"]])
    ttk.Label(bar2, text="重錄範圍：").pack(side="left")
    redo_cb = ttk.Combobox(bar2, textvariable=redo_var, values=REDO_CHOICES, state="readonly", width=40)
    redo_cb.pack(side="left")
    ttk.Label(bar2, text=f"　版本紀錄：{LOG_NAME}（↻＝檔案已存在，會被新版覆蓋）", foreground="#555").pack(side="left")

    def on_redo(_e=None):
        if S["running"]:
            redo_var.set(REDO_CHOICES[STATE["redo"]])
            return
        STATE["redo"] = REDO_CHOICES.index(redo_var.get())
        reload_plan()

    redo_cb.bind("<<ComboboxSelected>>", on_redo)

    ENG_CHOICES = ["Azure Speech（動用點數）", "edge-tts（免費，目前常 403）"]
    eng_var = tk.StringVar(value=ENG_CHOICES[0 if ENGINE == "azure" else 1])
    ttk.Label(bar2, text="　引擎：").pack(side="left")
    eng_cb = ttk.Combobox(bar2, textvariable=eng_var, values=ENG_CHOICES, state="readonly", width=26)
    eng_cb.pack(side="left")

    def on_eng(_e=None):
        global ENGINE
        if S["running"]:
            eng_var.set(ENG_CHOICES[0 if ENGINE == "azure" else 1])
            return
        ENGINE = "azure" if eng_var.get() == ENG_CHOICES[0] else "edge-tts"
        log(f"引擎改為：{ENGINE}" + ("　" + credit_line(root) if ENGINE == "azure" else ""))
        reload_plan()

    eng_cb.bind("<<ComboboxSelected>>", on_eng)

    reload_plan()
    log(f"目前引擎：{ENGINE}" + ("　" + credit_line(root) if ENGINE == "azure" else ""))
    if not S["groups"]:
        log("✔ 所有音檔都齊了，沒有需要產生的檔案。")
    poll()
    win.mainloop()
    return S["result"]


def part_status(root: Path, spec: Spec):
    """題庫選擇用：回傳 (可用?, 說明文字)。"""
    if not (root / spec.file).exists():

def parse_selection(ans, groups):
    ans = ans.strip().lower()
    if ans in ("all", "y", "全部"):
        return list(range(len(groups)))
    m = re.fullmatch(r"(?:前|first)\s*(\d+)", ans)
    if m:
        return list(range(min(int(m.group(1)), len(groups))))
    chosen = []
    for tok in re.split(r"[,\s，]+", ans):
        if not tok:
            continue
        m = re.fullmatch(r"(\d+)-(\d+)", tok)
        if m:
            chosen += [i for i in range(int(m.group(1)) - 1, int(m.group(2))) if 0 <= i < len(groups)]
        elif tok.isdigit():
            if 0 <= int(tok) - 1 < len(groups):
                chosen.append(int(tok) - 1)
        else:
            chosen += [i for i, g in enumerate(groups) if match_key(g["key"], tok)]
    return sorted(set(chosen))

# -*- coding: utf-8 -*-
"""
daily_timestamps.py — 為 TOEIC Daily Coach 的 mp3 產生「句子時間戳記」並寫入 daily.json

使用方式：雙擊執行 → 選擇 mp3（可複選，檔名如 w1d1.mp3）→ 選擇 daily.json
          也可由 json_merge.py 的「執行對時」帶參數開啟：daily_timestamps.py --json daily.json a.mp3 b.mp3
          （有帶參數就不再跳出選檔視窗，直接處理）
原理：用 faster-whisper 聽出每個字的時間，再與 daily.json 裡該天的 passage 對齊，
      只在對應那一天新增／更新 "timing" 欄位，其他資料一律不動。
【逐字對時】timing 每一句多一個 "words"：[[字元起, 字元迄, 開始秒, 結束秒], ...]，語音練習的語調曲線靠它把每個字釘在曲線上。
  舊資料（沒有 words）重新執行一次就會自動補上。
第一次執行會自動安裝 faster-whisper 並下載語音模型（約 150MB，需連網）。

【語調曲線】除了 timing，也會順便把整個 mp3 的「音高（F0）＋音量」算出來，寫進同一天的 "prosody" 欄位，
  網頁「語音練習 → 一句一句」會畫成曲線給你看（高低起伏＝語調、底下的柱狀＝輕重）。
  只用 numpy 計算，不需要另外安裝東西；4 分鐘的音檔約幾秒。
  每個選到的 mp3 會自己判斷要做什麼：
    ・還沒有 timing                → 聽寫對時（慢）＋算曲線
    ・已有 timing、但沒有曲線／曲線是舊版 → 只補曲線，不重新聽寫（很快）← 補舊檔就是選舊的 mp3 再跑一次
    ・mp3 檔案跟上次算曲線時不一樣（重錄過）→ 視為音檔換了，重新對時＋重算曲線
    ・已經都有、也沒換過           → 略過
  想強制全部重新對時：命令列加 --retime（或把下方 RETIME_EXISTING 改成 True）。

【判斷標準】固定使用下方 CFG（不需要每次選）。想調整才改 CFG 的數字；語音練習的「寬／嚴」在網頁裡切換，跟這裡無關。
"""
import sys, os, re, json, subprocess, difflib, traceback

MODEL = "base.en"   # 想更準可改 "small.en"（較慢、模型較大）
RETIME_EXISTING = False   # True＝已經有 timing 的檔案也一律重新聽寫對時（跟舊版一樣）；False＝有 timing 就只補曲線

# 一句話要同時滿足這些條件才算「對到」並寫入 timing；沒過的句子會在報告顯示 ✗，維護總覽會顯示「部分」
#   min_ratio：這句的字有多少比例在辨識結果裡找得到（舊版是 0.3，太鬆）
#   fuzzy    ：辨識聽成不同字時，相似度達多少才算（None＝不算，0＝一律算）
#   edges    ："any"＝句首或句尾至少一個字要對到（否則 start／end 會偏）；"both"＝兩個都要；None＝不檢查
#   max_rate ：每秒最多幾個字，超過代表時間被壓縮（正常約 2.5–4）；None＝不檢查（超過 3 個字的句子才檢查，太短的句子不準）
#   fill_gaps：True＝對不到的句子，若前後句都對到了且中間空檔合理，就依字數「推估」時間補上（報告顯示 ≈）；False＝不補
CFG = dict(min_ratio=0.60, fuzzy=0.70, edges="any", max_rate=8.0, fill_gaps=True)
ABBR = {"mr", "ms", "mrs", "dr", "inc", "co", "ltd", "no", "vs", "st", "jr", "sr"}


def norm(w):
    return re.sub(r"[^a-z0-9]", "", w.lower())


DIG = "zero one two three four five six seven eight nine".split()
SPEAKER = re.compile(r"[A-Za-z][A-Za-z.\-]*:")   # （舊）單字標籤，保留相容
# 說話者標籤（行首；與網站 speak.js 的 SK_LAB 同規則）：Customer: / The man : / Woman 1: / Mr. Lee:
LABEL = re.compile(r"^[ \t]*[A-Z][A-Za-z0-9.'’\-]*(?:[ \t]+[A-Za-z0-9.'’\-]+){0,2}[ \t]*[:：](?=\s|$)", re.M)


def label_spans(text):
    return [(m.start(), m.end()) for m in LABEL.finditer(text)]


def expand_words(words):
    """whisper 把數字寫成 64182 / 6-4-1-8-2 時，展開成 six four one eight two（時間平均分配）。"""
    out = []
    for s, e, t in words:
        n = norm(t)
        if not n:
            continue
        if n.isdigit():
            step = (e - s) / len(n)
            out += [(s + k * step, s + (k + 1) * step, DIG[int(c)]) for k, c in enumerate(n)]
        else:
            out.append((s, e, n))
    return out


def split_sentences(text):
    """回傳 [(from, to)]，為 passage 中每個句子的字元範圍。"""
    res, pos = [], 0

    def add(base, a, b, line):
        seg = line[a:b]
        lead = len(seg) - len(seg.lstrip())
        seg = seg.strip()
        if seg:
            res.append((base + a + lead, base + a + lead + len(seg)))

    for line in text.split("\n"):
        seg_start = 0
        for m in re.finditer(r'(?<=[.!?])\s+(?=[A-Z"\'(])', line):
            words = line[seg_start:m.start()].split()
            last = words[-1] if words else ""
            if last.rstrip(".").lower() in ABBR or re.fullmatch(r"(?:[A-Za-z]\.){2,}", last):
                continue  # A.M. / P.M. / Ms. 等縮寫不切句
            add(pos, seg_start, m.start(), line)
            seg_start = m.end()
        add(pos, seg_start, len(line), line)
        pos += len(line) + 1
    return res


def word_times(ptoks, idx, match, wtoks, start, end):
    """這一句每個字的時間：[[字元起, 字元迄, 開始秒, 結束秒], ...]（字元位置＝passage 內的絕對位置）。
    有對到的字用 whisper 聽到的時間；沒對到的字，在前後有時間的字之間依字母數平均分配。"""
    n = len(idx)
    ts = [None] * n
    for k, i in enumerate(idx):
        if i in match:
            w = wtoks[match[i]]
            ts[k] = [w[0], w[1]]
    k = 0
    while k < n:
        if ts[k] is not None:
            k += 1
            continue
        j = k
        while j < n and ts[j] is None:
            j += 1
        t0 = ts[k - 1][1] if k > 0 else start
        t1 = ts[j][0] if j < n else end
        if t1 < t0:
            t1 = t0
        wts = [max(1, len(ptoks[idx[q]][2])) for q in range(k, j)]
        tot, cur = float(sum(wts)), t0
        for q, wt in zip(range(k, j), wts):
            d = (t1 - t0) * wt / tot
            ts[q] = [cur, cur + d]
            cur += d
        k = j
    out = []
    for k, i in enumerate(idx):          # 數字被展開成好幾個字時（64182），合併回同一個字
        a, b = ptoks[i][0], ptoks[i][1]
        if out and out[-1][0] == a and out[-1][1] == b:
            out[-1][3] = max(out[-1][3], ts[k][1])
        else:
            out.append([a, b, ts[k][0], ts[k][1]])
    prev = 0.0
    for w in out:                        # 時間要由小到大，且每個字至少 0.05 秒
        w[2] = max(w[2], prev)
        w[3] = max(w[3], w[2] + 0.05)
        prev = w[2]
        w[2], w[3] = round(w[2], 2), round(w[3], 2)
    return out


def align(passage, words, cfg):
    """words: [(start, end, text)]（whisper 逐字結果）→ (timing 清單, 報告, 統計)。cfg＝判斷標準（見上方 CFG）。"""
    ptoks = []
    for m in re.finditer(r"\S+", passage):
        n = norm(m.group())
        if not n:
            continue
        if n.isdigit():   # 64182 → six four one eight two（與 whisper 結果的 expand_words 對稱，數字才對得上）
            ptoks += [(m.start(), m.end(), DIG[int(c)]) for c in n]
        else:
            ptoks.append((m.start(), m.end(), n))
    # 說話者標籤（行首的 Customer: / The man : / Woman 1:）音檔不會唸，不列入比對（句子範圍 from/to 不變）
    _ls = label_spans(passage)
    ptoks = [t for t in ptoks if not any(a <= t[0] < b for a, b in _ls)]
    wtoks = expand_words(words)
    pw, ww = [t[2] for t in ptoks], [t[2] for t in wtoks]
    sm = difflib.SequenceMatcher(None, pw, ww, autojunk=False)
    match = {}
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == "equal":
            for k in range(i2 - i1):
                match[i1 + k] = j1 + k
        elif tag == "replace" and i2 - i1 == j2 - j1 and cfg["fuzzy"] is not None:
            for k in range(i2 - i1):  # 聽成不一樣的字：依相似度決定算不算
                if difflib.SequenceMatcher(None, pw[i1 + k], ww[j1 + k]).ratio() >= cfg["fuzzy"]:
                    match[i1 + k] = j1 + k
    timing, report, sents = [], [], split_sentences(passage)
    res = []   # 每句：[a, b, 字數, label, timing 或 None, 失敗原因]
    for a, b in sents:
        idx = [i for i, t in enumerate(ptoks) if a <= t[0] < b]
        hit = [wtoks[match[i]] for i in idx if i in match]
        label = passage[a:b].replace("\n", " ")[:50]
        if not idx or not hit:
            res.append([a, b, max(len(idx), 1), label, None, "未對到（音檔可能沒唸這句）"])
            continue
        ratio = len(hit) / len(idx)
        start = round(min(h[0] for h in hit), 2)
        end = round(max(h[1] for h in hit), 2)
        end = max(end, start + 0.1)
        why = None
        if ratio < cfg["min_ratio"]:
            why = f"只對到 {ratio:.0%}（需 {cfg['min_ratio']:.0%}）"
        elif cfg["edges"] == "both" and not (idx[0] in match and idx[-1] in match):
            why = "句首或句尾的字沒對到"
        elif cfg["edges"] == "any" and idx[0] not in match and idx[-1] not in match:
            why = "句首與句尾都沒對到"
        elif cfg["max_rate"] and len(idx) > 3 and len(idx) / (end - start) > cfg["max_rate"]:
            why = f"語速 {len(idx) / (end - start):.1f} 字/秒太快，時間可能被壓縮"
        if why:
            res.append([a, b, len(idx), label, None, why])
            continue
        res.append([a, b, len(idx), label, {"start": start, "end": end, "from": a, "to": b,
                                             "words": word_times(ptoks, idx, match, wtoks, start, end)}, f"對中 {ratio:3.0%}"])

    # 補洞：連續幾句對不到、但前後句都對到時，依字數把中間的空檔分配給它們（推估值）
    if cfg.get("fill_gaps"):
        i = 0
        while i < len(res):
            if res[i][4] is not None:
                i += 1
                continue
            j = i
            while j < len(res) and res[j][4] is None:
                j += 1
            if 0 < i and j < len(res):
                t0, t1 = res[i - 1][4]["end"], res[j][4]["start"]
                n = sum(r[2] for r in res[i:j])
                gap = t1 - t0
                if gap > 0 and 1.0 <= n / gap <= 6.0:   # 空檔長度要像「真的有人唸了這幾句」
                    cur = t0
                    for r in res[i:j]:
                        d = gap * r[2] / n
                        r[4] = {"start": round(cur, 2), "end": round(max(cur + d, cur + 0.1), 2), "from": r[0], "to": r[1]}
                        r[5] = "≈推估（前後句夾住）"
                        cur += d
            i = j

    for a, b, n, label, t, note in res:
        if t is None:
            report.append(f"  ✗ {note}：{label}")
        else:
            timing.append(t)
            mark = "≈" if note.startswith("≈") else "✓"
            report.append(f"  {mark} {t['start']:6.2f}s – {t['end']:6.2f}s  {note}  {label}")
    return timing, report, len(sents)


# ───────────────────────── 文稿比對：以音檔為準，找出文稿與錄音不一致的地方 ─────────────────────────
# 用途：文稿定稿、錄完音後，又把文稿的字句刪掉或改掉（例如把 "The man :" 刪了再錄），
#       這裡用 whisper 聽到的內容當「事實」，和 daily.json 的 passage 逐字比對，列出每一處差異，
#       由人一筆一筆決定要不要改文稿（diff_passage → 畫面勾選 → apply_hunks → 重新對時 align）。
ALIAS = {"mister": "mr", "missus": "mrs", "doctor": "dr", "okay": "ok", "ms": "ms", "miss": "ms"}
_NUMW = set(DIG) | set((
    "ten eleven twelve thirteen fourteen fifteen sixteen seventeen eighteen nineteen twenty thirty forty fifty sixty "
    "seventy eighty ninety hundred thousand million and oh point dollar dollars percent am pm first second third "
    "fourth fifth sixth seventh eighth ninth tenth").split())


def _canon(w):
    return ALIAS.get(w, w)


def _is_numberish(ws):
    return bool(ws) and all(w.isdigit() or w in _NUMW for w in ws)


def _expand_passage(passage):
    """passage → [(a, b, 字)]；數字拆成英文數字（64182 → six four one eight two），每個字都對應原本那個詞的範圍。"""
    ls = []
    for m in re.finditer(r"\S+", passage):
        n = norm(m.group())
        if not n:
            continue
        if n.isdigit():
            ls += [(m.start(), m.end(), DIG[int(c)]) for c in n]
        else:
            ls.append((m.start(), m.end(), _canon(n)))
    return ls


def _expand_asr(words):
    """whisper 逐字 → [(start, end, 字, 原字)]；數字展開，原字只放在第一個。"""
    out = []
    for s, e, t in words:
        raw = (t or "").strip()
        n = norm(raw)
        if not n:
            continue
        if n.isdigit():
            step = (e - s) / len(n)
            out += [(s + k * step, s + (k + 1) * step, DIG[int(c)], raw if k == 0 else "") for k, c in enumerate(n)]
        else:
            out.append((s, e, _canon(n), raw))
    return out


def _strip_edge(w):
    return re.sub(r"^[^A-Za-z0-9]+|[^A-Za-z0-9]+$", "", w)


def diff_passage(passage, words, fuzzy=0.70):
    """回傳 (hunks, 略過的數字寫法差異數)。每個 hunk：
       {id, kind: delete|insert|replace|label, a, b, old, new, t0, t1, conf: sure|maybe, note}
       a/b＝這處差異在 passage 的字元範圍（insert 的 a==b＝插入位置）。"""
    pt, wt = _expand_passage(passage), _expand_asr(words)
    pw, ww = [t[2] for t in pt], [t[2] for t in wt]
    sm = difflib.SequenceMatcher(None, pw, ww, autojunk=False)
    lspans = label_spans(passage)
    raw_ops = []     # (i1, i2, j1, j2)
    skipped = 0
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == "equal":
            continue
        P, W = pw[i1:i2], ww[j1:j2]
        if tag == "replace":
            if "".join(P) == "".join(W):                      # check-in ↔ check in
                continue
            if _is_numberish(P) and _is_numberish(W):         # 25 ↔ twenty five：寫法不同，不是真的差異
                skipped += 1
                continue
            if len(P) == len(W):                              # 一對一：聽成相近的字不算差異，只留真的不同的連續段
                bad = [difflib.SequenceMatcher(None, a, b).ratio() < fuzzy for a, b in zip(P, W)]
                k = 0
                while k < len(P):
                    if not bad[k]:
                        k += 1
                        continue
                    m = k
                    while m < len(P) and bad[m]:
                        m += 1
                    raw_ops.append((i1 + k, i1 + m, j1 + k, j1 + m))
                    k = m
                continue
        if tag == "delete" and _is_numberish(P):
            skipped += 1
            continue
        raw_ops.append((i1, i2, j1, j2))

    hunks = []
    for n, (i1, i2, j1, j2) in enumerate(raw_ops, 1):
        # 音檔時間窗：前一個對到的字的結束 ～ 後一個對到的字的開始
        prev_end = wt[j1 - 1][1] if j1 > 0 else 0.0
        next_start = wt[j2][0] if j2 < len(wt) else (wt[-1][1] if wt else 0.0)
        h = {"id": n, "t0": round(prev_end, 2), "t1": round(max(next_start, prev_end), 2), "conf": "maybe", "note": ""}
        if i2 > i1:
            a, b = pt[i1][0], pt[i2 - 1][1]
            h["old"] = passage[a:b]
        else:
            # 純插入：放在前一個字之後（沒有前字就放在第一個字之前）
            a = b = pt[i1 - 1][1] if i1 > 0 else (pt[0][0] if pt else 0)
            h["old"] = ""
        h["a"], h["b"] = a, b
        raw_new = [t[3] for t in wt[j1:j2] if t[3]]
        h["new"] = " ".join(_strip_edge(w) for w in raw_new if _strip_edge(w))
        gap = h["t1"] - h["t0"]
        if i2 > i1 and j2 == j1:
            h["kind"] = "delete"
            in_label = any(la <= a and b <= lb + 2 for la, lb in lspans) or any(la <= a < lb for la, lb in lspans)
            if in_label:
                h["kind"] = "label"
            need = 0.15 + 0.12 * (i2 - i1)
            if passage[:a].rstrip(" \t").endswith("\n") or not passage[:a].strip():
                need += 0.8          # 行首的字（多半是說話者標籤）：前面本來就會有換人說話的停頓
            if gap <= need:
                h["conf"], h["note"] = "sure", "音檔這裡只有 %.1f 秒空檔，確實沒唸" % gap
            else:
                h["note"] = "音檔這裡有 %.1f 秒的聲音，可能是辨識漏聽，請聽一下再決定" % gap
        elif i2 == i1:
            h["kind"] = "insert"
            at_edge = j1 == 0 or j2 >= len(wt)
            h["conf"] = "maybe"
            h["note"] = "音檔開頭／結尾多出來的字，常是辨識亂加（如 Thank you）" if at_edge else "音檔有唸、文稿沒有"
        else:
            h["kind"] = "replace"
            h["note"] = "文稿與音檔用字不同（音檔 %.1f 秒）" % gap
        hunks.append(h)
    return hunks, skipped


def _cut_span(text, a, b):
    """刪掉 text[a:b]，順手清掉殘留的冒號、多餘空白，並把新的句首字母改大寫。回傳新字串。"""
    m = re.match(r"[ \t]*[:：\-–—]+(?=\s|$)", text[b:])
    if m:
        b += m.end()
    before, after = text[:a], text[b:]
    at_line = before == "" or before.rstrip(" \t").endswith("\n") or before.rstrip(" \t") == ""
    if at_line:
        before = before.rstrip(" \t")
        after = after.lstrip(" \t")
    elif before.endswith((" ", "\t")) and after.startswith((" ", "\t")):
        after = after.lstrip(" \t")
    elif not before.endswith((" ", "\t")) and after.startswith((" ", "\t")) and after.lstrip(" \t")[:1] in ".,!?;:":
        after = after.lstrip(" \t")
    if before.endswith((" ", "\t")) and after.startswith("\n"):
        before = before.rstrip(" \t")
    if before.endswith("\n") and after.startswith("\n"):     # 整行刪空了 → 不留空行
        after = after[1:]
    # 句首大寫
    pre = before.rstrip(" \t")
    if pre == "" or pre.endswith(("\n", ".", "!", "?", '"', "”", "“")):
        mm = re.match(r"^([\"“'(]*)([a-z])", after)
        if mm:
            after = after[:mm.start(2)] + after[mm.start(2)].upper() + after[mm.end(2):]
    return before + after


def apply_hunks(passage, hunks):
    """把勾選的 hunks 套用到 passage，回傳新文稿。由後往前套，字元位置才不會跑掉。"""
    text = passage
    for h in sorted(hunks, key=lambda x: (x["a"], x["b"]), reverse=True):
        a, b, kind = h["a"], h["b"], h["kind"]
        if kind in ("delete", "label"):
            text = _cut_span(text, a, b)
        elif kind == "insert":
            ins = h["new"]
            if not ins:
                continue
            nxt = text[a:a + 1]
            if a == 0 or (b == a and text[:a].strip() == ""):
                text = ins + " " + text[a:]
            else:
                text = text[:a] + " " + ins + text[a:]
        elif kind == "replace":
            old = text[a:b]
            lead = re.match(r"^[^A-Za-z0-9]*", old).group()
            trail = re.search(r"[^A-Za-z0-9]*$", old).group()
            body = h["new"]
            if body:
                if old[len(lead):len(lead) + 1].isupper() and body[:1].islower():
                    body = body[0].upper() + body[1:]
                text = text[:a] + lead + body + trail + text[b:]
            else:
                text = _cut_span(text, a, b)
    return text


def transcribe_cached(model_getter, mp3, cache_dir, prompt=None, log=print):
    """辨識結果存成快取（backup/asr_cache/<檔名>.json）：同一個 mp3 沒改過就不用再聽一次。
       model_getter＝呼叫後回傳 whisper 模型的函式（沒有快取時才會載入模型）。"""
    os.makedirs(cache_dir, exist_ok=True)
    st = os.stat(mp3)
    key = {"model": MODEL, "size": st.st_size, "mtime": int(st.st_mtime)}
    cp = os.path.join(cache_dir, os.path.splitext(os.path.basename(mp3))[0] + ".json")
    try:
        with open(cp, encoding="utf-8") as f:
            c = json.load(f)
        if c.get("key") == key and isinstance(c.get("words"), list):
            log("使用快取的辨識結果：%s" % os.path.basename(cp))
            return [tuple(w) for w in c["words"]]
    except Exception:
        pass
    words = transcribe(model_getter(), mp3, prompt)
    with open(cp, "w", encoding="utf-8") as f:
        json.dump({"key": key, "words": words}, f, ensure_ascii=False)
    return words


# ───────────────────────── 語調曲線：音高（F0）＋音量，整個 mp3 一條線 ─────────────────────────
# 寫進 daily.json 該天的 "prosody"：{"v":1,"hop":0.05,"src":<mp3 檔案大小>,"f0":[Hz…],"vol":[0–100…]}
#   f0 ：每 hop 秒一個點，單位 Hz（整數），0＝沒有音高（無聲、氣音、停頓）
#   vol：每 hop 秒一個點，0–100（以這個音檔最大聲的地方當 100，往下 50 dB 當 0）
#   src：mp3 的檔案大小；mp3 換掉（重錄）後大小不同 → 下次執行會知道曲線過期，自動重算
#   網頁依 timing 的 start/end 從這條線切出每一句，所以句子被合併、剪裁微調都不用重算。
PROSODY_VER = 1
P_SR, P_HOP, P_WIN = 16000, 800, 1024      # 取樣率；每 800 點（0.05 秒）一格；每格看 1024 點（64ms）
P_FMIN, P_FMAX = 65.0, 420.0               # 找音高的範圍（男聲約 85–180、女聲約 165–255）
P_VOICED = 0.38                            # 週期性要有多明顯才算「有音高」（0–1，越高越嚴格）
P_GATE_DB = 40.0                           # 比最大聲的地方小聲超過這麼多 dB 的格子，一律當無聲


def _p_decode(mp3):
    """mp3 → 16kHz 單聲道 float 陣列。先用 faster-whisper 內建的解碼；它失敗（av 版本不合）就改用 av，再不行用 ffmpeg 指令。"""
    import numpy as np
    ensure_faster_whisper()
    try:
        from faster_whisper.audio import decode_audio
        return decode_audio(mp3, sampling_rate=P_SR)
    except Exception:
        pass
    try:
        import av
        rs = av.AudioResampler(format="s16", layout="mono", rate=P_SR)
        chunks = []
        with av.open(mp3, mode="r") as c:
            for fr in c.decode(audio=0):
                for o in rs.resample(fr):
                    chunks.append(o.to_ndarray().reshape(-1))
            for o in rs.resample(None):
                chunks.append(o.to_ndarray().reshape(-1))
        return np.concatenate(chunks).astype(np.float32) / 32768.0
    except Exception:
        pass
    raw = subprocess.run(["ffmpeg", "-v", "error", "-i", mp3, "-f", "s16le", "-ac", "1", "-ar", str(P_SR), "-"],
                         check=True, stdout=subprocess.PIPE).stdout
    return np.frombuffer(raw, dtype=np.int16).astype(np.float32) / 32768.0


def analyze_prosody(mp3, log=print):
    """讀 mp3 → 回傳 prosody 字典。只用 numpy（faster-whisper 已經裝了），不需要 librosa。"""
    import numpy as np
    y = np.asarray(_p_decode(mp3), dtype=np.float64)
    n = max(1, int(np.ceil(len(y) / P_HOP)))
    pad = np.zeros(len(y) + P_WIN)
    pad[P_WIN // 2: P_WIN // 2 + len(y)] = y            # 第 k 格以 k*HOP 為中心
    win = np.hanning(P_WIN)
    # 視窗本身的自相關：用來修正「看越遠的 lag 重疊越少」造成的偏差
    nf = 2 * P_WIN
    rw = np.fft.irfft(np.abs(np.fft.rfft(win, nf)) ** 2)[:P_WIN]
    rw = rw / rw[0]
    lmin, lmax = int(P_SR / P_FMAX), int(P_SR / P_FMIN)
    rms = np.zeros(n)
    f0 = np.zeros(n)
    strength = np.zeros(n)
    frames = np.lib.stride_tricks.sliding_window_view(pad, P_WIN)[::P_HOP][:n]
    rms[:len(frames)] = np.sqrt(np.mean(frames ** 2, axis=1))
    db = 20 * np.log10(rms + 1e-9)
    top = np.percentile(db[rms > 1e-6], 98) if np.any(rms > 1e-6) else 0.0
    for k in range(len(frames)):
        if db[k] < top - P_GATE_DB:
            continue
        x = frames[k] - frames[k].mean()
        x = x * win
        r0 = float(np.dot(x, x))
        if r0 <= 1e-12:
            continue
        r = np.fft.irfft(np.abs(np.fft.rfft(x, nf)) ** 2)[:P_WIN]
        r = r / r0 / np.maximum(rw, 1e-3)
        seg = r[lmin - 1: lmax + 2]
        # 區域最大值當候選；取「夠高又最短的週期」避免抓成低八度
        pk = [i for i in range(1, len(seg) - 1) if seg[i] > seg[i - 1] and seg[i] >= seg[i + 1] and seg[i] > P_VOICED]
        if not pk:
            continue
        best = max(seg[i] for i in pk)
        i0 = next(i for i in pk if seg[i] >= 0.88 * best)
        a, b, c = seg[i0 - 1], seg[i0], seg[i0 + 1]
        den = a - 2 * b + c
        off = 0.5 * (a - c) / den if abs(den) > 1e-12 else 0.0
        lag = (lmin - 1) + i0 + off
        if lag > 0:
            f0[k] = P_SR / lag
            strength[k] = b
    f0 = _p_clean(f0)
    v = np.clip(100 + (db - top) * 100.0 / 50.0, 0, 100)
    v[rms < 1e-6] = 0
    return {"v": PROSODY_VER, "hop": P_HOP / P_SR,
            "src": os.path.getsize(mp3),
            "f0": [int(round(x)) for x in f0], "vol": [int(round(x)) for x in v]}


def _p_clean(f0):
    """去掉單獨亂跳的點、修正突然差八度的點、再做 5 點中位數平滑（只在有音高的連續段內做）。"""
    import numpy as np
    f = f0.copy()
    n = len(f)
    i = 0
    while i < n:                       # 一連串有音高的段落
        if f[i] <= 0:
            i += 1
            continue
        j = i
        while j < n and f[j] > 0:
            j += 1
        if j - i < 3:                  # 太短（不到 0.15 秒）多半是雜訊
            f[i:j] = 0
        else:
            seg = f[i:j].copy()
            med = float(np.median(seg))
            for t in range(len(seg)):  # 與整段中位數差八度 → 折回來
                while seg[t] > med * 1.65 and seg[t] / 2 >= P_FMIN:
                    seg[t] /= 2
                while seg[t] < med / 1.65 and seg[t] * 2 <= P_FMAX:
                    seg[t] *= 2
            sm = seg.copy()
            for t in range(len(seg)):
                lo, hi = max(0, t - 2), min(len(seg), t + 3)
                sm[t] = np.median(seg[lo:hi])
            f[i:j] = sm
        i = j
    return f


def dump_json(data):
    text = json.dumps(data, ensure_ascii=False, indent=2)
    # 讓每個 timing 物件維持單行，避免 JSON 檔變得很長
    text = re.sub(
        r'\{\s*"start": ([\d.]+),\s*"end": ([\d.]+),\s*"from": (\d+),\s*"to": (\d+)\s*\}',
        r'{"start": \1, "end": \2, "from": \3, "to": \4}', text)
    # 語調曲線的數字陣列（f0／vol）也壓成一行，不然一個數字一行檔案會暴增
    return re.sub(r'("(?:f0|vol)": )\[([^\]]*)\]',
                  lambda m: m.group(1) + "[" + re.sub(r"\s+", "", m.group(2)) + "]", text)

def transcribe(model, mp3, prompt=None):
    print(f"辨識中：{os.path.basename(mp3)}")
    segs, info = model.transcribe(mp3, language="en", word_timestamps=True,
                                  beam_size=5, condition_on_previous_text=False,
                                  initial_prompt=prompt or None)
    words = []
    for seg in segs:
        print(f"  [{seg.start:6.1f}s] {seg.text.strip()[:60]}")
        words += [(w.start, w.end, w.word) for w in (seg.words or [])]
    return words


def fresh_prosody(entry, mp3):
    """這天已經有、而且是用目前這個 mp3 算的最新版曲線？"""
    pr = entry.get("prosody")
    return (isinstance(pr, dict) and pr.get("v") == PROSODY_VER and bool(pr.get("f0"))
            and pr.get("src") == os.path.getsize(mp3))

            if day == 8 and m.group(3):          # D8 專案：w1d8-2 ＝ Week 1 Day 9（d8-N ＝ day 7+N，和 json_merge.py、網頁一致）
                day = 7 + int(m.group(3))
        elif len(mp3s) == 1:
            # 只選一個檔案時，才跳出視窗讓你手動輸入
            week = simpledialog.askinteger("週次", "檔名不是 w1d1 格式，請輸入 Week（例如 5）：")
            day = simpledialog.askinteger("天數", "請輸入 Day（1-7）：")
            if week is None or day is None:
                print("✗ 未輸入週次／天數，略過。")
                failed.append(f"{name}（未輸入週次／天數）")
                continue
        else:
            print("✗ 檔名不是 w1d1 格式，略過。")
            failed.append(f"{name}（檔名不是 w1d1 格式）")
            continue

        entry = next((x for x in items if isinstance(x, dict)
                      and x.get("week") == week and x.get("day") == day), None)
        if entry is None:
            print(f"✗ daily.json 裡沒有 Week {week} Day {day}，略過。")
            failed.append(f"{name}（找不到 Week {week} Day {day}）")
            continue
        print(f"目標：Week {week} Day {day}｜{entry.get('tag', '')}")

        # 這個檔案要做什麼？
        old = entry.get("prosody")
        changed = isinstance(old, dict) and old.get("src") not in (None, os.path.getsize(mp3))   # 曲線是用別的 mp3 算的 → 音檔換過
        no_words = bool(entry.get("timing")) and any(isinstance(t, dict) and not t.get("words") for t in entry["timing"]
                                                   if t.get("end", 0) - t.get("start", 0) > 0.0)   # 舊版對時沒有逐字時間
        need_time = retime or not entry.get("timing") or changed or no_words
        need_curve = need_time or not fresh_prosody(entry, mp3)
        if not need_curve:
            print("✓ 已經有 timing 和語調曲線，略過（要重新對時請加 --retime）。")
            skipped.append(name)
            continue
        if no_words and not retime and not changed:
            print("※ 這一天的 timing 還沒有逐字時間（舊版對時），重新聽寫一次補上。")
        if changed and not retime and entry.get("timing"):
            print("※ 這個 mp3 跟上次算曲線時不一樣（重錄過？），重新對時並重算曲線。")

        timing = None
        if need_time:
            try:
                if model is None:
                    model = load_model(MODEL)
                timing, report, total = align(entry["passage"], transcribe(model, mp3), CFG)
            except Exception as e:
                traceback.print_exc()
                print(f"✗ 處理時發生錯誤，略過。")
                failed.append(f"{name}（錯誤：{e}）")
                continue
            print("對齊結果：")
            print("\n".join(report))
            if not timing:
                print("✗ 沒有任何句子對得上，略過（請確認 mp3 內容與英文文稿一致）。")
                failed.append(f"{name}（沒有句子對得上）")
                continue
        else:
            print("已有 timing，只補語調曲線（不需要聽寫）。")

        old_t = entry.get("timing") or []
        if timing is not None and old_t and len(timing) < len(old_t) and not retime:
            # 新結果比原本少很多句：多半是 mp3 和這一天的文稿不是同一篇。不覆蓋，原本的 timing 與曲線都保留
            print(f"✗ 這次只對到 {len(timing)} 句，原本有 {len(old_t)} 句；為了不蓋掉原本的資料，略過這個檔案。")
            print("  （檔名對應到 Week %d Day %d，請確認 mp3 檔名；真的要覆蓋請加 --retime）" % (week, day))
            failed.append(f"{name}（只對到 {len(timing)}/{len(old_t)} 句，已保留原本的 timing；檔名是否對應到正確的那一天？）")
            continue

        try:
            print("計算語調曲線（音高＋音量）…")
            pros = analyze_prosody(mp3)
            voiced = sum(1 for v in pros["f0"] if v > 0)
            print(f"  曲線 {len(pros['f0'])} 點（{len(pros['f0']) * pros['hop']:.1f} 秒），有音高的佔 {voiced / max(1, len(pros['f0'])):.0%}")
        except Exception as e:
            traceback.print_exc()
            print("⚠ 語調曲線計算失敗（timing 仍會照常寫入）。")
            failed.append(f"{name}（語調曲線失敗：{e}）")
            pros = None

        if timing is not None:
            entry["timing"] = timing
            n_time += 1
            print(f"✓ Week {week} Day {day}：{len(timing)} / {total} 個句子")
            if len(timing) < total:
                partial.append(f"{name}（{len(timing)}/{total}）")
        if pros is not None:
            entry["prosody"] = pros
            n_curve += 1
        elif timing is not None:
            entry.pop("prosody", None)   # 音檔換了但曲線算不出來：不要留舊的

    if n_time == 0 and n_curve == 0:
        if failed:
            msg = "所有檔案都處理失敗，未寫入任何資料。\n\n" + "\n".join(failed)
            print("\n" + msg)
            return messagebox.showerror("沒有可寫入的資料", msg)
        msg = "選的檔案都已經有 timing 和語調曲線，沒有需要更新的。\n（要重新對時：命令列加 --retime）"
        print("\n" + msg)
        return messagebox.showinfo("沒有變更", msg)

    tmp = jpath + ".tmp"
    with open(tmp, "w", encoding="utf-8") as f:
        f.write(dump_json(data) + "\n")
    os.replace(tmp, jpath)

    msg = f"完成！對時 {n_time} 個、語調曲線 {n_curve} 個（模型：{MODEL}）。"
    if skipped:
        msg += f"\n略過 {len(skipped)} 個（已經完成）。"
    if partial:
        msg += "\n\n有漏句（維護總覽會顯示「部分」），可改用較大的模型（MODEL 改 small.en）或放寬 CFG 再跑一次：\n" + "\n".join(partial)
    if failed:
        msg += "\n\n失敗／略過：\n" + "\n".join(failed)
    print("\n" + msg)
    messagebox.showinfo("完成", msg)


if __name__ == "__main__":
    try:
        main()
    except Exception:
        traceback.print_exc()
    input("\n按 Enter 關閉視窗…")
